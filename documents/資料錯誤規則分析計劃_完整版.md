## 五、分析執行計劃

### 5.1 Phase 1：基礎統計（優先執行）

**目標**：建立各欄位的統計基線

| 步驟 | 分析內容 | 方法 | 輸出 | 預估時間 |
|------|---------|------|------|---------|
| 1.1 | 空值率統計 | Python Pandas | null_analysis.md | 30 分鐘 |
| 1.2 | 值分布統計 | Python Pandas | value_distribution.md | 1 小時 |
| 1.3 | 數值範圍建立 | IQR 計算 | numeric_baselines.yaml | 30 分鐘 |

**1.1 空值率分析腳本**:

```python
def analyze_null_rates(df, table_name):
    """分析各欄位空值率"""
    total = len(df)
    results = []

    for col in df.columns:
        if col.startswith('_'):
            continue
        null_count = df[col].isna().sum()
        empty_count = (df[col] == '').sum()
        missing = null_count + empty_count
        rate = round(missing / total * 100, 2)

        results.append({
            'table': table_name,
            'column': col,
            'null_count': null_count,
            'empty_count': empty_count,
            'missing_rate': rate,
            'severity': 'HIGH' if rate > 10 else 'MEDIUM' if rate > 5 else 'LOW'
        })

    return results
```

```python
import re

def parse_logistics_name(name):
    """解析物流名稱"""
    if not name:
        return {'error': 'empty'}

    # 主要模式: 廠商_溫層_發貨點_取貨點_運費方式
    pattern = r'^(.+?)_(.+?)_(.+?)_(.+?)_(.+?)$'
    match = re.match(pattern, str(name))

    if match:
        return {
            'vendor': match.group(1),
            'temperature': match.group(2),
            'origin': match.group(3),
            'destination': match.group(4),
            'fee_type': match.group(5),
            'parsed': True
        }

    # 次要模式: 廠商_溫層_發貨點
    pattern2 = r'^(.+?)_(.+?)_(.+?)$'
    match2 = re.match(pattern2, str(name))

    if match2:
        return {
            'vendor': match2.group(1),
            'temperature': match2.group(2),
            'origin': match2.group(3),
            'parsed': True,
            'partial': True
        }

    return {'raw': name, 'parsed': False}
```

```python
def check_foreign_key(df_main, df_ref, main_col, ref_col, name):
    """檢查外鍵完整性"""
    # 取得參照表的有效值
    valid_values = set(df_ref[ref_col].dropna().unique())

    # 檢查主表中不存在的值
    main_values = df_main[main_col].dropna()
    missing = main_values[~main_values.isin(valid_values)]

    return {
        'name': name,
        'total_records': len(df_main),
        'non_null_records': len(main_values),
        'missing_count': len(missing),
        'missing_rate': round(len(missing) / len(main_values) * 100, 2) if len(main_values) > 0 else 0,
        'missing_values': missing.unique().tolist()[:20]  # 取前 20 個
    }
```

```python
def learn_association_rules(df, col_a, col_b, min_count=1):
    """從歷史資料學習關聯規則"""
    # 過濾空值
    valid = df[[col_a, col_b]].dropna()

    # 統計組合出現次數
    combo_stats = valid.groupby([col_a, col_b]).size().reset_index(name='count')

    # 按 col_a 分組
    rules = {}
    for a_val, group in combo_stats.groupby(col_a):
        rules[a_val] = {
            'allowed_values': group[col_b].tolist(),
            'counts': dict(zip(group[col_b], group['count'])),
            'total': group['count'].sum()
        }

    return rules
```

```python
def build_numeric_baseline(df, column):
    """建立數值欄位的統計基線"""
    values = pd.to_numeric(df[column], errors='coerce').dropna()

    if len(values) == 0:
        return None

    Q1 = values.quantile(0.25)
    Q3 = values.quantile(0.75)
    IQR = Q3 - Q1

    return {
        'column': column,
        'count': len(values),
        'min': float(values.min()),
        'max': float(values.max()),
        'mean': float(values.mean()),
        'median': float(values.median()),
        'Q1': float(Q1),
        'Q3': float(Q3),
        'IQR': float(IQR),
        'lower_bound': float(Q1 - 1.5 * IQR),
        'upper_bound': float(Q3 + 1.5 * IQR),
        'outlier_count': int(((values < Q1 - 1.5 * IQR) | (values > Q3 + 1.5 * IQR)).sum())
    }
```
