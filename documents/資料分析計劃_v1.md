```python
def ai_batch_analyze(candidates: list, batch_size: int = 50):
    """
    AI 分批分析候選異常

    Args:
        candidates: 候選異常列表（已精簡欄位）
        batch_size: 每批處理筆數（預設 50）

    Returns:
        分析結果列表
    """
    results = []
    total_batches = (len(candidates) + batch_size - 1) // batch_size

    for i in range(0, len(candidates), batch_size):
        batch = candidates[i:i + batch_size]
        batch_num = i // batch_size + 1

        print(f"處理批次 {batch_num}/{total_batches}...")

        # 呼叫 AI 分析（每批獨立 Context）
        result = analyze_batch(batch)
        results.extend(result)

    return results
```

### 4.1 問題 1：物流欄位混合

**現況**：
```
物流名稱範例：「蝦皮新竹_常溫_總倉_宅配_自付運費」
```

**分析方法**：Python 正則解析

```python
import re

def parse_logistics_name(name: str) -> dict:
    """
    解析物流名稱為獨立維度

    範例輸入: "蝦皮新竹_常溫_總倉_宅配_自付運費"
    範例輸出: {
        "vendor": "蝦皮新竹",
        "temperature": "常溫",
        "origin": "總倉",
        "destination": "宅配",
        "fee_type": "自付運費"
    }
    """
    pattern = r"^(.+?)_(.+?)_(.+?)_(.+?)_(.+?)$"
    match = re.match(pattern, name)
    if match:
        return {
            "vendor": match.group(1),
            "temperature": match.group(2),
            "origin": match.group(3),
            "destination": match.group(4),
            "fee_type": match.group(5)
        }
    return {"raw": name, "parse_error": True}
```

**預期輸出**：
- 物流名稱解析規則
- 無法解析的異常名稱清單
- 各維度值分布統計

```python
def analyze_null_rates(df: pd.DataFrame, table_name: str) -> dict:
    """
    分析各欄位空值率
    """
    total = len(df)
    null_stats = {}

    for col in df.columns:
        null_count = df[col].isna().sum()
        empty_count = (df[col] == '').sum()
        total_missing = null_count + empty_count

        null_stats[col] = {
            "null_count": null_count,
            "empty_count": empty_count,
            "total_missing": total_missing,
            "missing_rate": round(total_missing / total * 100, 2)
        }

    return null_stats
```

### 4.3 問題 3：商品結構/系列選錯

**分析方法**：交叉分析 + 規則檢查

```python
def check_product_structure_consistency(df: pd.DataFrame) -> list:
    """
    檢查商品結構與其他欄位的一致性
    """
    anomalies = []

    for idx, row in df.iterrows():
        structure = row.get('商品結構')
        content = row.get('商品內容', '')
        name = row.get('商品名稱', '')

        # 規則 1: 結構=組合，但內容只有單一項目
        if structure == '組合' and ',' not in content and '+' not in content:
            anomalies.append({
                "_ragicId": row['_ragicId'],
                "rule": "P1",
                "desc": "結構為組合但內容似乎是單品",
                "structure": structure,
                "content": content
            })

        # 規則 2: 結構=單品，但名稱包含組合關鍵字
        if structure == '單品' and any(kw in name for kw in ['組合', '套裝', '禮盒']):
            anomalies.append({
                "_ragicId": row['_ragicId'],
                "rule": "P2",
                "desc": "結構為單品但名稱包含組合關鍵字",
                "structure": structure,
                "name": name
            })

    return anomalies
```

```python
def check_quantity_anomalies(df: pd.DataFrame) -> list:
    """
    檢查數量欄位異常
    """
    anomalies = []

    # 數量 <= 0
    zero_or_negative = df[df['數量'] <= 0]

    # 數量異常大（IQR 方法）
    Q1 = df['數量'].quantile(0.25)
    Q3 = df['數量'].quantile(0.75)
    IQR = Q3 - Q1
    upper_bound = Q3 + 1.5 * IQR

    outliers = df[df['數量'] > upper_bound]

    return {
        "zero_or_negative": len(zero_or_negative),
        "outliers": len(outliers),
        "upper_bound": upper_bound
    }
```

### 4.5 問題 5：電商平臺買家

**分析方法**：模式匹配

```python
PLATFORM_PATTERNS = [
    ('MOMO', r'(?i)momo|富邦媒體'),
    ('蝦皮', r'(?i)shopee|蝦皮'),
    ('PCHome', r'(?i)pchome|網家'),
    ('Yahoo', r'(?i)yahoo|雅虎'),
    ('樂天', r'(?i)rakuten|樂天'),
]

def identify_platform_customers(df: pd.DataFrame) -> dict:
    """
    識別平臺買家
    """
    results = {platform: 0 for platform, _ in PLATFORM_PATTERNS}
    results['一般客戶'] = 0

    platform_records = []

    for idx, row in df.iterrows():
        name = row.get('客戶名稱', '')
        postal = row.get('郵遞區號', '')

        matched = False
        for platform, pattern in PLATFORM_PATTERNS:
            if re.search(pattern, name) or postal == '999':
                results[platform] += 1
                platform_records.append({
                    "_ragicId": row['_ragicId'],
                    "customer_name": name,
                    "platform": platform
                })
                matched = True
                break

        if not matched:
            results['一般客戶'] += 1

    return {
        "summary": results,
        "records": platform_records
    }
```

### 5.3 關聯規則學習

**目標**：學習「品牌-促銷」「品牌-商品」等組合的合法關係

```python
def learn_brand_promotion_rules(df_99: pd.DataFrame) -> dict:
    """
    從訂單明細學習品牌-促銷關聯規則

    輸出格式:
    {
        "CYT": {  # 品牌編號
            "allowed_promotions": ["MK001", "MK002", "MK003"],
            "promotion_counts": {"MK001": 1500, "MK002": 800},
            "first_seen": "2023-01-15",
            "last_seen": "2024-12-09"
        },
        ...
    }
    """
    # 只保留有效記錄（品牌和促銷都有值）
    valid = df_99[
        df_99['品牌編號'].notna() &
        df_99['通路活動號碼1'].notna()
    ]

    # 統計品牌-促銷組合
    combo_stats = valid.groupby(['品牌編號', '通路活動號碼1']).agg({
        '_ragicId': 'count',
        '訂單日期': ['min', 'max']
    }).reset_index()

    combo_stats.columns = ['brand', 'promotion', 'count', 'first_seen', 'last_seen']

    # 轉換為規則格式
    rules = {}
    for brand, group in combo_stats.groupby('brand'):
        rules[brand] = {
            'allowed_promotions': group['promotion'].tolist(),
            'promotion_counts': dict(zip(group['promotion'], group['count'])),
            'first_seen': str(group['first_seen'].min()),
            'last_seen': str(group['last_seen'].max())
        }

    return rules
```

### 5.4 數值基線學習

**目標**：建立各數值欄位的合理範圍

```python
def learn_numeric_baselines(df: pd.DataFrame, numeric_columns: list) -> dict:
    """
    學習數值欄位的統計基線

    輸出格式:
    {
        "訂單實收": {
            "Q1": 500, "Q3": 3500, "IQR": 3000,
            "lower_bound": -4000,  # Q1 - 1.5*IQR
            "upper_bound": 8000,   # Q3 + 1.5*IQR
            "mean": 2100, "std": 1500,
            "z_lower": -2400, "z_upper": 6600
        },
        ...
    }
    """
    baselines = {}

    for col in numeric_columns:
        values = pd.to_numeric(df[col], errors='coerce').dropna()
        if len(values) == 0:
            continue

        Q1 = values.quantile(0.25)
        Q3 = values.quantile(0.75)
        IQR = Q3 - Q1
        mean = values.mean()
        std = values.std()

        baselines[col] = {
            'Q1': float(Q1), 'Q3': float(Q3), 'IQR': float(IQR),
            'lower_bound': float(Q1 - 1.5 * IQR),
            'upper_bound': float(Q3 + 1.5 * IQR),
            'mean': float(mean), 'std': float(std),
            'z_lower': float(mean - 3 * std),
            'z_upper': float(mean + 3 * std),
            'sample_size': len(values)
        }

    return baselines
```

```python
def learn_phone_pattern(df: pd.DataFrame, column: str) -> dict:
    """從成功樣本學習電話格式模式"""
    phones = df[column].dropna().astype(str)

    patterns = {
        'mobile_09': r'^09\d{8}$',
        'mobile_dash': r'^09\d{2}-\d{3}-\d{3}$',
        'landline': r'^0\d{1,2}-?\d{7,8}$',
        'intl': r'^\+886\d{9,10}$'
    }

    pattern_counts = {}
    for name, pattern in patterns.items():
        matches = phones.str.match(pattern).sum()
        pattern_counts[name] = int(matches)

    dominant = max(pattern_counts, key=pattern_counts.get)

    return {
        'column': column,
        'dominant_pattern': patterns[dominant],
        'pattern_distribution': pattern_counts,
        'valid_rate': sum(pattern_counts.values()) / len(phones)
    }
```

```python
def learn_valid_combinations(df_99: pd.DataFrame) -> dict:
    """學習多欄位的合法組合"""
    combinations = {}

    # 品牌-通路組合
    brand_channel = df_99.groupby(['品牌編號', '通路編號']).size()
    combinations['brand_channel'] = {
        'pairs': [
            {'brand': b, 'channel': c, 'count': int(cnt)}
            for (b, c), cnt in brand_channel.items()
        ],
        'total_combinations': len(brand_channel)
    }

    # 通路-金流組合
    channel_payment = df_99.groupby(['通路編號', '金流編號']).size()
    combinations['channel_payment'] = {
        'pairs': [
            {'channel': ch, 'payment': p, 'count': int(cnt)}
            for (ch, p), cnt in channel_payment.items()
        ]
    }

    # 通路-物流組合
    channel_logistics = df_99.groupby(['通路編號', '物流編號']).size()
    combinations['channel_logistics'] = {
        'pairs': [
            {'channel': ch, 'logistics': l, 'count': int(cnt)}
            for (ch, l), cnt in channel_logistics.items()
        ]
    }

    return combinations
```

```python
class CleaningRuleApplier:
    """清理規則應用器"""

    def __init__(self, rules_dir: str):
        self.association_rules = self._load_yaml(f'{rules_dir}/association_rules.yaml')
        self.numeric_baselines = self._load_yaml(f'{rules_dir}/numeric_baselines.yaml')
        self.pattern_rules = self._load_yaml(f'{rules_dir}/pattern_rules.yaml')
        self.valid_combinations = self._load_yaml(f'{rules_dir}/valid_combinations.yaml')

    def apply_all(self, record: dict) -> list:
        """
        對單筆記錄應用所有規則

        Returns:
            list of violations: [{rule_id, severity, message}]
        """
        violations = []
        violations.extend(self._check_associations(record))
        violations.extend(self._check_numeric_bounds(record))
        violations.extend(self._check_patterns(record))
        violations.extend(self._check_combinations(record))
        return violations

    def _check_associations(self, record: dict) -> list:
        """檢查關聯規則"""
        violations = []
        brand = record.get('品牌編號')
        promotion = record.get('通路活動號碼1')

        if brand and promotion:
            brand_rules = self.association_rules.get(brand, {})
            allowed = brand_rules.get('allowed_promotions', [])

            if allowed and promotion not in allowed:
                violations.append({
                    'rule_id': 'A1',
                    'rule_type': 'association',
                    'severity': 'high',
                    'message': f'品牌 {brand} 從未使用過促銷 {promotion}',
                    'fields': ['品牌編號', '通路活動號碼1'],
                    'values': [brand, promotion]
                })

        return violations
```
