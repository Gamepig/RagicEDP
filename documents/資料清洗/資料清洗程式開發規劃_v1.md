## 一、系統概述

### 1.1 目標

建立自動化資料清洗系統，在增量備份完成後：
1. 從 BigQuery 提取當日新增資料
2. 執行 SQL 規則清洗與欄位補足
3. 使用 AI 判斷無法自動修正的問題
4. 標記需人工處理的資料
5. 通知相關人員進行修正
6. 提供 UI 介面進行人工修正
7. 顯示星狀模型圖供分析使用

### 1.2 設計原則

| 原則 | 說明 |
|------|------|
| **增量處理** | 只處理當日新增資料，不重複清洗已處理資料 |
| **原始保留** | 保留原始值，清洗結果存於獨立欄位 |
| **可追溯** | 所有修改記錄完整歷史 |
| **漸進式** | SQL 優先，AI 輔助，人工兜底 |

### 1.3 開發注意事項

> **重要**: 資料清洗是在增量備份**完成之後**執行，不是同時運行。
>
> 執行順序：
> 1. Cloud Scheduler 每日 00:00 觸發增量備份
> 2. 增量備份完成後，觸發資料清洗流程
> 3. 清洗完成後，發送通知

#### 3.1.1 清洗引擎 (`engine.py`)

```python
"""資料清洗引擎"""
from dataclasses import dataclass
from datetime import datetime
from typing import Dict, List, Any, Optional
from enum import Enum

class CleaningStatus(Enum):
    """清洗狀態"""
    PENDING = "pending"           # 待處理
    AUTO_FIXED = "auto_fixed"     # 自動修正
    AI_FIXED = "ai_fixed"         # AI 修正
    MANUAL_REQUIRED = "manual"    # 需人工處理
    COMPLETED = "completed"       # 已完成
    SKIPPED = "skipped"           # 略過

@dataclass
class CleaningResult:
    """清洗結果"""
    record_id: str                # 記錄 ID (_ragicId)
    table_code: str               # 表格代碼
    original_values: Dict         # 原始值
    fixed_values: Dict            # 修正後值
    issues: List[Dict]            # 發現的問題
    status: CleaningStatus        # 處理狀態
    ai_suggestion: Optional[str]  # AI 建議
    confidence_score: float       # 信心度 (0-1)
    cleaned_at: datetime          # 清洗時間
    cleaned_by: str               # 清洗者 (system/ai/user)


class CleaningEngine:
    """清洗引擎主類別"""

    def __init__(self, bq_client, ai_analyzer=None):
        self.bq_client = bq_client
        self.ai_analyzer = ai_analyzer
        self.sql_cleaner = SQLCleaner(bq_client)
        self.rule_loader = RuleLoader()
        self.field_fixer = FieldFixer()
        self.result_writer = ResultWriter(bq_client)

    def run_daily_cleaning(self, backup_date: datetime = None) -> Dict[str, Any]:
        """
        執行每日清洗

        Args:
            backup_date: 備份日期（預設今天）

        Returns:
            清洗結果摘要
        """
        if backup_date is None:
            backup_date = datetime.now().date()

        results = {
            'date': backup_date.isoformat(),
            'tables': {},
            'total_processed': 0,
            'auto_fixed': 0,
            'ai_fixed': 0,
            'manual_required': 0,
            'errors': []
        }

        # 依序處理各表
        for table_code in TABLE_PRIORITY_ORDER:
            table_result = self._clean_table(table_code, backup_date)
            results['tables'][table_code] = table_result

            # 累計統計
            results['total_processed'] += table_result['processed']
            results['auto_fixed'] += table_result['auto_fixed']
            results['ai_fixed'] += table_result['ai_fixed']
            results['manual_required'] += table_result['manual_required']

        return results

    def _clean_table(self, table_code: str, backup_date: datetime) -> Dict:
        """清洗單一表格"""
        # Step 1: 取得當日新增資料
        new_records = self._fetch_new_records(table_code, backup_date)

        if not new_records:
            return {'status': 'skipped', 'reason': 'no_new_records'}

        cleaning_results = []

        for record in new_records:
            # Step 2: SQL 規則檢查
            issues = self.sql_cleaner.check_record(table_code, record)

            if not issues:
                # 無問題，直接標記完成
                result = CleaningResult(
                    record_id=record['_ragicId'],
                    table_code=table_code,
                    original_values=record,
                    fixed_values=record,
                    issues=[],
                    status=CleaningStatus.COMPLETED,
                    ai_suggestion=None,
                    confidence_score=1.0,
                    cleaned_at=datetime.now(),
                    cleaned_by='system'
                )
            else:
                # Step 3: 嘗試自動修正
                fixed_record, remaining_issues = self.field_fixer.auto_fix(
                    table_code, record, issues
                )

                if not remaining_issues:
                    # 全部自動修正成功
                    result = CleaningResult(
                        record_id=record['_ragicId'],
                        table_code=table_code,
                        original_values=record,
                        fixed_values=fixed_record,
                        issues=issues,
                        status=CleaningStatus.AUTO_FIXED,
                        ai_suggestion=None,
                        confidence_score=0.95,
                        cleaned_at=datetime.now(),
                        cleaned_by='system'
                    )
                else:
                    # Step 4: AI 判斷
                    if self.ai_analyzer:
                        ai_result = self.ai_analyzer.analyze(
                            table_code, fixed_record, remaining_issues
                        )

                        if ai_result['can_fix'] and ai_result['confidence'] >= 0.8:
                            # AI 修正
                            result = CleaningResult(
                                record_id=record['_ragicId'],
                                table_code=table_code,
                                original_values=record,
                                fixed_values=ai_result['fixed_record'],
                                issues=issues,
                                status=CleaningStatus.AI_FIXED,
                                ai_suggestion=ai_result['suggestion'],
                                confidence_score=ai_result['confidence'],
                                cleaned_at=datetime.now(),
                                cleaned_by='ai'
                            )
                        else:
                            # 需人工處理
                            result = CleaningResult(
                                record_id=record['_ragicId'],
                                table_code=table_code,
                                original_values=record,
                                fixed_values=fixed_record,
                                issues=remaining_issues,
                                status=CleaningStatus.MANUAL_REQUIRED,
                                ai_suggestion=ai_result.get('suggestion'),
                                confidence_score=ai_result.get('confidence', 0),
                                cleaned_at=datetime.now(),
                                cleaned_by='pending'
                            )
                    else:
                        # 無 AI，直接標記人工
                        result = CleaningResult(
                            record_id=record['_ragicId'],
                            table_code=table_code,
                            original_values=record,
                            fixed_values=fixed_record,
                            issues=remaining_issues,
                            status=CleaningStatus.MANUAL_REQUIRED,
                            ai_suggestion=None,
                            confidence_score=0,
                            cleaned_at=datetime.now(),
                            cleaned_by='pending'
                        )

            cleaning_results.append(result)

        # Step 5: 寫入結果
        self.result_writer.write_results(cleaning_results)

        return self._summarize_results(cleaning_results)

    def _fetch_new_records(self, table_code: str, backup_date: datetime) -> List[Dict]:
        """取得當日新增記錄"""
        table_name = TABLE_MAPPING[table_code]
        query = f"""
            SELECT *
            FROM `{self.project}.{self.dataset}.{table_name}`
            WHERE DATE(_backup_date) = '{backup_date.isoformat()}'
              AND (_cleaning_status IS NULL OR _cleaning_status = 'pending')
        """
        return list(self.bq_client.query(query).result())
```

#### 3.1.2 SQL 規則執行器 (`sql_cleaner.py`)

```python
"""SQL 規則執行器"""
from typing import Dict, List, Any
import re


class SQLCleaner:
    """SQL 清洗規則執行器"""

    def __init__(self, bq_client):
        self.bq_client = bq_client
        self.rules = {}  # 規則快取

    def check_record(self, table_code: str, record: Dict) -> List[Dict]:
        """
        檢查單筆記錄是否符合規則

        Returns:
            問題清單 [{'rule_id': 'FMT-001', 'field': '行動電話', 'issue': '...'}]
        """
        issues = []
        rules = self._get_rules_for_table(table_code)

        for rule in rules:
            issue = self._check_rule(rule, record)
            if issue:
                issues.append(issue)

        return issues

    def _check_rule(self, rule: Dict, record: Dict) -> Optional[Dict]:
        """執行單一規則檢查"""
        rule_type = rule['type']

        if rule_type == 'format':
            return self._check_format(rule, record)
        elif rule_type == 'required':
            return self._check_required(rule, record)
        elif rule_type == 'range':
            return self._check_range(rule, record)
        elif rule_type == 'foreign_key':
            return self._check_fk(rule, record)
        elif rule_type == 'unique':
            return self._check_unique(rule, record)

        return None

    def _check_format(self, rule: Dict, record: Dict) -> Optional[Dict]:
        """格式驗證"""
        field = rule['field']
        value = record.get(field)

        if value is None or value == '':
            return None  # 空值由 required 規則處理

        pattern = rule['pattern']
        if not re.match(pattern, str(value)):
            return {
                'rule_id': rule['id'],
                'rule_type': 'format',
                'field': field,
                'value': value,
                'issue': f"格式不符: 應符合 {rule.get('description', pattern)}",
                'severity': rule.get('severity', 'medium'),
                'auto_fixable': rule.get('auto_fixable', False)
            }

        return None

    def _check_required(self, rule: Dict, record: Dict) -> Optional[Dict]:
        """必填欄位檢查"""
        field = rule['field']
        value = record.get(field)

        if value is None or value == '':
            return {
                'rule_id': rule['id'],
                'rule_type': 'required',
                'field': field,
                'value': None,
                'issue': f"必填欄位為空",
                'severity': rule.get('severity', 'high'),
                'auto_fixable': False
            }

        return None

    def _check_range(self, rule: Dict, record: Dict) -> Optional[Dict]:
        """數值範圍檢查"""
        field = rule['field']
        value = record.get(field)

        if value is None:
            return None

        try:
            num_value = float(value)
            min_val = rule.get('min')
            max_val = rule.get('max')

            if min_val is not None and num_value < min_val:
                return {
                    'rule_id': rule['id'],
                    'rule_type': 'range',
                    'field': field,
                    'value': value,
                    'issue': f"數值低於最小值 {min_val}",
                    'severity': rule.get('severity', 'medium'),
                    'auto_fixable': False
                }

            if max_val is not None and num_value > max_val:
                return {
                    'rule_id': rule['id'],
                    'rule_type': 'range',
                    'field': field,
                    'value': value,
                    'issue': f"數值超過最大值 {max_val}",
                    'severity': rule.get('severity', 'medium'),
                    'auto_fixable': False
                }
        except (ValueError, TypeError):
            return {
                'rule_id': rule['id'],
                'rule_type': 'range',
                'field': field,
                'value': value,
                'issue': "無法解析為數值",
                'severity': 'high',
                'auto_fixable': False
            }

        return None

    def _check_fk(self, rule: Dict, record: Dict) -> Optional[Dict]:
        """外鍵參照檢查"""
        field = rule['field']
        value = record.get(field)

        if value is None or value == '':
            return None

        ref_table = rule['ref_table']
        ref_field = rule['ref_field']

        # 查詢參照表
        query = f"""
            SELECT 1 FROM `{self.project}.{self.dataset}.{ref_table}`
            WHERE {ref_field} = @value
            LIMIT 1
        """
        job_config = bigquery.QueryJobConfig(
            query_parameters=[
                bigquery.ScalarQueryParameter("value", "STRING", str(value))
            ]
        )

        result = list(self.bq_client.query(query, job_config=job_config).result())

        if not result:
            return {
                'rule_id': rule['id'],
                'rule_type': 'foreign_key',
                'field': field,
                'value': value,
                'issue': f"參照不存在: {ref_table}.{ref_field}",
                'severity': 'high',
                'auto_fixable': False
            }

        return None
```

#### 3.1.3 欄位自動修正器 (`field_fixer.py`)

```python
"""欄位自動修正器"""
from typing import Dict, List, Tuple
import re


class FieldFixer:
    """自動修正可修正的欄位問題"""

    # 自動修正規則
    FIX_RULES = {
        'FMT-002-FIX': '_fix_landline_missing_zero',
        'FMT-004-FIX': '_fix_tax_id_padding',
        'FMT-007-FIX': '_fix_date_format',
    }

    def auto_fix(
        self,
        table_code: str,
        record: Dict,
        issues: List[Dict]
    ) -> Tuple[Dict, List[Dict]]:
        """
        嘗試自動修正問題

        Returns:
            (修正後記錄, 剩餘問題清單)
        """
        fixed_record = record.copy()
        remaining_issues = []

        for issue in issues:
            if issue.get('auto_fixable'):
                rule_id = issue['rule_id']
                fix_method = self.FIX_RULES.get(rule_id)

                if fix_method and hasattr(self, fix_method):
                    try:
                        fixed_value = getattr(self, fix_method)(
                            issue['field'],
                            issue['value']
                        )
                        if fixed_value is not None:
                            fixed_record[issue['field']] = fixed_value
                            continue  # 修正成功，不加入剩餘問題
                    except Exception as e:
                        issue['fix_error'] = str(e)

            remaining_issues.append(issue)

        return fixed_record, remaining_issues

    def _fix_landline_missing_zero(self, field: str, value: str) -> Optional[str]:
        """修正市話缺少開頭 0"""
        if not value:
            return None

        digits = re.sub(r'[^\d]', '', str(value))

        # 2 開頭 + 9 碼 → 台北市話缺少 0
        if digits.startswith('2') and len(digits) == 9:
            return '0' + digits

        # 3-8 開頭 + 8 碼 → 其他縣市市話缺少 0
        if digits[0] in '345678' and len(digits) == 8:
            return '0' + digits

        return None

    def _fix_tax_id_padding(self, field: str, value: str) -> Optional[str]:
        """修正統一編號補零"""
        if not value:
            return None

        digits = re.sub(r'[^\d]', '', str(value))

        # 7 碼補零
        if len(digits) == 7:
            return '0' + digits

        return None

    def _fix_date_format(self, field: str, value: str) -> Optional[str]:
        """修正日期格式"""
        if not value:
            return None

        # 嘗試各種格式轉換為標準格式 YYYY/MM/DD
        patterns = [
            (r'^(\d{4})-(\d{2})-(\d{2})$', r'\1/\2/\3'),     # YYYY-MM-DD
            (r'^(\d{2})/(\d{2})/(\d{4})$', r'\3/\1/\2'),     # MM/DD/YYYY
            (r'^(\d{4})(\d{2})(\d{2})$', r'\1/\2/\3'),       # YYYYMMDD
        ]

        for pattern, replacement in patterns:
            if re.match(pattern, str(value)):
                return re.sub(pattern, replacement, str(value))

        return None
```

```python
        分析問題資料並提供修正建議

                'can_fix': bool,
                'confidence': float,
                'fixed_record': Dict,
                'suggestion': str,

            }

            f"  {k}: {v}"
```

```python
import requests
from typing import Optional


class LineSender:
    """LINE Notify 發送器"""

    def __init__(self, token: str = None):
        self.token = token or os.getenv('LINE_NOTIFY_TOKEN')
        self.api_url = "https://notify-api.line.me/api/notify"

    def send(self, message: str, image_url: Optional[str] = None) -> bool:
        """
        發送 LINE 通知

        Args:
            message: 訊息內容
            image_url: 可選的圖片 URL

        Returns:
            是否發送成功
        """
        headers = {
            "Authorization": f"Bearer {self.token}"
        }

        data = {
            "message": message
        }

        if image_url:
            data["imageThumbnail"] = image_url
            data["imageFullsize"] = image_url

        response = requests.post(
            self.api_url,
            headers=headers,
            data=data
        )

        return response.status_code == 200
```

```python
"""星狀模型生成器"""
from typing import Dict, List, Any
from google.cloud import bigquery


class StarSchemaGenerator:
    """星狀模型圖生成器"""

    # 表格定義
    FACT_TABLES = ['fact_orders', 'fact_order_details']

    DIM_TABLES = {
        'dim_brand': {'pk': '品牌編號', 'display': '品牌表'},
        'dim_channel': {'pk': '通路編號', 'display': '通路表'},
        'dim_payment': {'pk': '金流編號', 'display': '金流表'},
        'dim_logistics': {'pk': '物流編號', 'display': '物流表'},
        'dim_postal': {'pk': '郵遞區號', 'display': '郵遞區號表'},
        'dim_customer': {'pk': '客戶編號', 'display': '客戶表'},
        'dim_product': {'pk': '商品編號', 'display': '商品表'},
        'dim_campaign': {'pk': '活動編號', 'display': '活動管理表'},
    }

    # 關聯定義
    RELATIONSHIPS = [
        {'from': 'fact_order_details', 'to': 'dim_brand', 'fk': '品牌編號'},
        {'from': 'fact_order_details', 'to': 'dim_channel', 'fk': '通路編號'},
        {'from': 'fact_order_details', 'to': 'dim_payment', 'fk': '金流編號'},
        {'from': 'fact_order_details', 'to': 'dim_logistics', 'fk': '物流編號'},
        {'from': 'fact_order_details', 'to': 'dim_postal', 'fk': '郵遞區號'},
        {'from': 'fact_order_details', 'to': 'dim_customer', 'fk': '客戶編號'},
        {'from': 'fact_order_details', 'to': 'dim_product', 'fk': '商品編號'},
        {'from': 'fact_order_details', 'to': 'fact_orders', 'fk': '訂單編號'},
        {'from': 'dim_product', 'to': 'dim_brand', 'fk': '品牌編號'},
    ]

    def __init__(self, bq_client, project_id: str, dataset: str):
        self.bq_client = bq_client
        self.project_id = project_id
        self.dataset = dataset

    def generate_mermaid(self, level: str = "overview") -> str:
        """
        生成 Mermaid ER 圖

        Args:
            level: "overview" | "detailed"

        Returns:
            Mermaid 程式碼
        """
        lines = ["erDiagram"]

        # 生成關聯線
        for rel in self.RELATIONSHIPS:
            lines.append(f'    {rel["to"]} ||--o{{ {rel["from"]} : "{rel["fk"]}"')

        lines.append("")

        # 生成實體定義
        if level == "overview":
            # 總覽模式：只顯示表名和主鍵
            for table, info in self.DIM_TABLES.items():
                lines.append(f"    {table} {{")
                lines.append(f'        STRING {info["pk"]} PK')
                lines.append("    }")
                lines.append("")

            # 事實表
            lines.append("    fact_orders {")
            lines.append("        STRING 訂單編號 PK")
            lines.append("        DATE 訂單日期")
            lines.append("    }")
            lines.append("")

            lines.append("    fact_order_details {")
            lines.append("        INT _ragicId PK")
            lines.append("        STRING 訂單編號 FK")
            lines.append("    }")

        elif level == "detailed":
            # 詳細模式：顯示所有欄位
            for table_name in list(self.DIM_TABLES.keys()) + self.FACT_TABLES:
                schema = self._get_table_schema(table_name)
                lines.append(f"    {table_name} {{")
                for field in schema[:10]:  # 限制顯示前 10 個欄位
                    field_type = self._map_bq_type(field.field_type)
                    pk_marker = " PK" if self._is_pk(table_name, field.name) else ""
                    fk_marker = " FK" if self._is_fk(table_name, field.name) else ""
                    lines.append(f"        {field_type} {field.name}{pk_marker}{fk_marker}")
                lines.append("    }")
                lines.append("")

        return "\n".join(lines)

    def generate_stats(self) -> Dict[str, Any]:
        """生成統計資訊"""
        stats = {
            'tables': {},
            'total_records': 0,
            'relationships': len(self.RELATIONSHIPS),
        }

        for table_name in list(self.DIM_TABLES.keys()) + self.FACT_TABLES:
            count = self._get_table_count(table_name)
            stats['tables'][table_name] = {
                'count': count,
                'type': 'dimension' if table_name.startswith('dim_') else 'fact',
            }
            stats['total_records'] += count

        return stats

    def _get_table_schema(self, table_name: str) -> List:
        """取得表格 Schema"""
        table_ref = f"{self.project_id}.{self.dataset}.{table_name}"
        table = self.bq_client.get_table(table_ref)
        return table.schema

    def _get_table_count(self, table_name: str) -> int:
        """取得表格記錄數"""
        query = f"SELECT COUNT(*) as cnt FROM `{self.project_id}.{self.dataset}.{table_name}`"
        result = list(self.bq_client.query(query).result())
        return result[0].cnt if result else 0

    def _map_bq_type(self, bq_type: str) -> str:
        """BigQuery 型別對應"""
        mapping = {
            'STRING': 'STRING',
            'INTEGER': 'INT64',
            'INT64': 'INT64',
            'FLOAT': 'FLOAT64',
            'FLOAT64': 'FLOAT64',
            'DATE': 'DATE',
            'DATETIME': 'DATETIME',
            'TIMESTAMP': 'TIMESTAMP',
            'BOOLEAN': 'BOOL',
            'BOOL': 'BOOL',
        }
        return mapping.get(bq_type, 'STRING')
```
